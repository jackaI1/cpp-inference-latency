cmake_minimum_required(VERSION 3.16)
project(inference_benchmark LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ONNXRUNTIME_ROOT "" CACHE PATH "ONNX Runtime release directory")
find_path(ORT_INCLUDE onnxruntime_cxx_api.h HINTS "${ONNXRUNTIME_ROOT}/include" REQUIRED)
find_library(ORT_LIBRARY onnxruntime HINTS "${ONNXRUNTIME_ROOT}/lib" REQUIRED)
add_executable(benchmark src/main.cpp)
target_include_directories(benchmark PRIVATE src "${ORT_INCLUDE}")
target_link_libraries(benchmark PRIVATE "${ORT_LIBRARY}")
enable_testing()
add_executable(stats_test tests/stats_test.cpp)
target_include_directories(stats_test PRIVATE src)
add_test(NAME stats COMMAND stats_test)
